\newcounter{cpcounter}
\newcommand{\dosdig}[1]{\ifnum\value{#1}<10 0\fi\arabic{#1}}

\newenvironment{pruebasde}[2]{%
  \subsection*{\color{leipAzul}#1\quad #2}
  \addcontentsline{toc}{subsection}{#1\quad #2}
  \vspace{-0.4em}
  {\color{leipGris}\hrule height 0.5pt}
  \smallskip
}{%
  \bigskip
}

\newenvironment{tablapruebas}{%
  \renewcommand{\arraystretch}{1.25}%
  \begin{longtable}{@{}p{1.2cm} >{\raggedright\arraybackslash}p{4.1cm} >{\raggedright\arraybackslash}p{4.3cm} >{\raggedright\arraybackslash}p{4.3cm}@{}}

  \textbf{C\'odigo} & \textbf{Nombre} & \textbf{Procedimiento} & \textbf{Resultado esperado} \\
  \midrule
  \endhead
}{%
  \bottomrule
  \end{longtable}
  \renewcommand{\arraystretch}{1}%
}
